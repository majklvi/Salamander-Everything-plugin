cmake_minimum_required(VERSION 3.21)
project(PrerequisiteRegression LANGUAGES CXX)
add_subdirectory(../.. plugin EXCLUDE_FROM_ALL)
add_executable(prerequisite_test main.cpp)
target_include_directories(prerequisite_test PRIVATE ../../src ../../third_party/everything-sdk-v2/include "${SAMANDARIN_SDK_DIR}")
target_compile_features(prerequisite_test PRIVATE cxx_std_17)
target_compile_definitions(prerequisite_test PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(prerequisite_test PRIVATE version)
enable_testing()
add_test(NAME missing_source COMMAND prerequisite_test)
